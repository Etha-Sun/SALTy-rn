Record Recordint32x4_t : Type := make_record_int32x4_t {
  int32x4_t_s0 : Z;
  int32x4_t_s1 : Z;
  int32x4_t_s2 : Z;
  int32x4_t_s3 : Z
}.

Definition zero_record_int32x4_t : Recordint32x4_t :=
  make_record_int32x4_t 0 0 0 0.

Definition StructPredint32x4_t (x : Z) (v : Recordint32x4_t) : Assertion :=
  ((&(x # "int32x4_t" ->ₛ "s0")) # Int  |-> v.(int32x4_t_s0)) **
  ((&(x # "int32x4_t" ->ₛ "s1")) # Int  |-> v.(int32x4_t_s1)) **
  ((&(x # "int32x4_t" ->ₛ "s2")) # Int  |-> v.(int32x4_t_s2)) **
  ((&(x # "int32x4_t" ->ₛ "s3")) # Int  |-> v.(int32x4_t_s3)).

Definition UndefStructPredint32x4_t (x : Z) : Assertion :=
  ((&(x # "int32x4_t" ->ₛ "s0")) # Int  |->_) **
  ((&(x # "int32x4_t" ->ₛ "s1")) # Int  |->_) **
  ((&(x # "int32x4_t" ->ₛ "s2")) # Int  |->_) **
  ((&(x # "int32x4_t" ->ₛ "s3")) # Int  |->_).

Lemma store_Predint32x4_t_to_undef : forall x v,
  StructPredint32x4_t x v |-- UndefStructPredint32x4_t x.
Proof.
  intros.
  unfold StructPredint32x4_t, UndefStructPredint32x4_t.
  sep_apply store_int_undef_store_int.
  sep_apply store_int_undef_store_int.
  sep_apply store_int_undef_store_int.
  sep_apply store_int_undef_store_int.
  cancel.
Qed.

Lemma undefstore_Predint32x4_t_to_align : forall x,
  UndefStructPredint32x4_t x |-- store_align_n ((((0 + sizeof(INT)) + sizeof(INT)) + sizeof(INT)) + sizeof(INT)).
Proof.
  intros.
  unfold UndefStructPredint32x4_t.
  sep_apply undef_store_int_align4.
  sep_apply undef_store_int_align4.
  sep_apply undef_store_int_align4.
  sep_apply undef_store_int_align4.
  sep_apply store_align4_to_store_align.
  sep_apply (store_align_merge (4 * 1)).
  sep_apply (store_align_merge ((4 * 1 + 4 * 1))).
  sep_apply (store_align_merge (((4 * 1 + 4 * 1) + 4 * 1))).
  try rewrite sizeof_int.
  try rewrite sizeof_int.
  try rewrite sizeof_int.
  try rewrite sizeof_int.
  replace (((((0 + 4) + 4) + 4) + 4)) with 16 by lia.
  replace ((((4 * 1 + 4 * 1) + 4 * 1) + 4 * 1)) with 16 by lia.
  cancel.
Qed.

Lemma store_Predint32x4_t_to_align : forall x v,
  StructPredint32x4_t x v |-- store_align_n ((((0 + sizeof(INT)) + sizeof(INT)) + sizeof(INT)) + sizeof(INT)).
Proof.
  intros.
  sep_apply store_Predint32x4_t_to_undef.
  sep_apply undefstore_Predint32x4_t_to_align.
  cancel.
Qed.

Module StoreStructint32x4_tAsElement <: ELEMENT_STORE.
  Definition A := Recordint32x4_t.
  Definition sizeA := ((((0 + sizeof(INT)) + sizeof(INT)) + sizeof(INT)) + sizeof(INT)).
  Definition storeA (x: addr) (lo: Z) (a: Recordint32x4_t): Assertion :=
    (StructPredint32x4_t (x + lo * sizeA) a).
  Definition undefstoreA (x: addr) (lo: Z): Assertion :=
    (UndefStructPredint32x4_t (x + lo * sizeA)).

  Lemma store_to_undefstore : forall x lo a,
    storeA x lo a |-- undefstoreA x lo.
  Proof.
    intros.
    unfold storeA, undefstoreA.
    sep_apply store_Predint32x4_t_to_undef.
    cancel.
  Qed.

  Lemma storeA_shift : forall x n lo a,
    storeA (x + n * sizeA) lo a --||-- storeA x (lo + n) a.
  Proof.
    intros.
    unfold storeA.
    replace (x + n * sizeA + lo * sizeA) with (x + (lo + n) * sizeA) by lia.
    reflexivity.
  Qed.

  Lemma undefstoreA_shift : forall x n lo,
    undefstoreA (x + n * sizeA) lo --||-- undefstoreA x (lo + n).
  Proof.
    intros.
    unfold undefstoreA.
    replace (x + n * sizeA + lo * sizeA) with (x + (lo + n) * sizeA) by lia.
    reflexivity.
  Qed.

  Lemma store_to_align : forall x lo a, storeA x lo a |-- store_align_n sizeA.
  Proof.
    intros.
    unfold storeA, sizeA.
    sep_apply store_Predint32x4_t_to_align.
    cancel.
  Qed.

  Lemma undefstore_to_align : forall x lo, undefstoreA x lo |-- store_align_n sizeA.
  Proof.
    intros.
    unfold undefstoreA, sizeA.
    sep_apply undefstore_Predint32x4_t_to_align.
    cancel.
  Qed.

  Lemma sizeA_valid : 0 < sizeA < Int.max_unsigned.
  Proof.
    unfold sizeA.
  try rewrite sizeof_int.
  try rewrite sizeof_int.
  try rewrite sizeof_int.
  try rewrite sizeof_int.
    replace Int.max_unsigned with 4294967295 by reflexivity.
    lia.
  Qed.

End StoreStructint32x4_tAsElement.

Module Structint32x4_tArray := ArrayLib (StoreStructint32x4_tAsElement).
