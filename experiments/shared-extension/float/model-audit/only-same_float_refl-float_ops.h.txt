#ifndef SHARED_FLOAT_H
#define SHARED_FLOAT_H
/*@
  logic float fp_add32(float a, float b);
  logic float f32_read{L}(float *p, integer k) = p[k];
  predicate same_value(float a, float b) =
    \eq_float(a,b) || (\is_NaN(a) && \is_NaN(b));
  predicate same_float(float a, float b) =
    (\is_NaN(a) && \is_NaN(b)) ||
    (same_value(a,b) && \sign(a) == \sign(b));
  lemma same_float_refl: \forall float a; same_float(a,a);
*/
/*@ ghost
/@ requires same_value(a,fp_add32(x,y));
    requires same_value(b,fp_add32(x,y));

    ensures same_value(a,b); @/
void proof_float_join(float a,float b,float x,float y) {}
*/
/*@ assigns \nothing; ensures same_value(\result, fp_add32(a,b)); */
float f32_add(float a, float b);
/*@ requires \valid_read(p+(0..3)); assigns \nothing; ensures \forall integer i; 0<=i<4 ==> same_float(\result.lane[i],p[i]); */
float32x4_t vld1q_f32(const float *p);
/*@ assigns \nothing; ensures \forall integer i; 0<=i<4 ==> same_value(\result.lane[i],fp_add32(a.lane[i],b.lane[i])); */
float32x4_t vaddq_f32(float32x4_t a, float32x4_t b);
/*@ requires \valid(p+(0..3)); assigns p[0..3]; ensures \forall integer i; 0<=i<4 ==> same_float(p[i],a.lane[i]); ensures \forall float *q; \valid_read(q) && \separated(q,p+(0..4-1)) ==> same_float(*q,\old(*q)); */
void vst1q_f32(float *p, float32x4_t a);
/*@ assigns \nothing; ensures \forall integer i; 0<=i<2 ==> same_float(\result.lane[i],a.lane[i]); */
float32x2_t vget_low_f32(float32x4_t a);
/*@ assigns \nothing; ensures \forall integer i; 0<=i<2 ==> same_float(\result.lane[i],a.lane[i+2]); */
float32x2_t vget_high_f32(float32x4_t a);
/*@ requires \valid(p+(0..1)); assigns p[0..1]; ensures \forall integer i; 0<=i<2 ==> same_float(p[i],a.lane[i]); ensures \forall float *q; \valid_read(q) && \separated(q,p+(0..2-1)) ==> same_float(*q,\old(*q)); */
void vst1_f32(float *p, float32x2_t a);
/*@ requires 0<=lane<2; requires \valid(p); assigns *p; ensures same_float(*p,a.lane[lane]); ensures \forall float *q; \valid_read(q) && \separated(q,p+(0..1-1)) ==> same_float(*q,\old(*q)); */
void vst1_lane_f32(float *p, float32x2_t a, int lane);
/*@ requires vl<=32; requires \valid_read(p+(0..vl-1)); assigns \nothing; ensures \forall integer i; 0<=i<vl ==> same_float(\result.lane[i],p[i]); */

/*@ requires vl<=32; assigns \nothing; ensures \forall integer i; 0<=i<vl ==> same_value(\result.lane[i],fp_add32(a.lane[i],b.lane[i])); */
vfloat32m8_t __riscv_vfadd_vv_f32m8(vfloat32m8_t a, vfloat32m8_t b, size_t vl);
/*@ requires vl<=32; requires \valid(p+(0..vl-1)); assigns p[0..vl-1]; ensures \forall integer i; 0<=i<vl ==> same_float(p[i],a.lane[i]); ensures \forall float *q; \valid_read(q) && \separated(q,p+(0..vl-1)) ==> same_float(*q,\old(*q)); */
void __riscv_vse32_v_f32m8(float *p, vfloat32m8_t a, size_t vl);
